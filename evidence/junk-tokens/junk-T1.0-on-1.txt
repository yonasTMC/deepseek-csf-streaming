def percentile(sorted_values: Sequence[float], p: float) -> float:
    if not sorted_values: raise ValueError...
    if not 0 <= p <= 100: raise ValueError...
    if len(sorted_values) == 1: return float(sorted_values[0])

    lower = math.floor(rank)

def percentile(sorted_values: Sequence[float], p: float) -> float:
    if not sorted_values:
        raise ValueError("percentile() requires at least one value")
    if not math.isfinite(p) or not 0 <= p <= 100:
        raise ValueError("percentile must be between 0 and 100")
    if len(sorted_values) == 1:
        return float(sorted_values[0])
    rank = (len(sorted_values) - 1) * (p / 100.0)
    lower = math.floor(rank)
    upper = math.ceil(rank)
    if lower == upper:
        return float(sorted_values[lower])
    fraction = rank - lower
    return float(sorted_values[lower] * (1.0 - fraction) + sorted_values[upper] * fraction)
```

def format_table(rows: Sequence[Mapping[str, object]], percentiles: Sequence[float]) -> str:
    if not rows:
        return "No matching log records."
    p_labels = [f"P{p:g}" for p in percentiles]
    headers = ["Endpoint", "Count", "Min", *p_labels, "Max"]
    table_rows: list[list[str]] = []
    for row in rows:
        values = [
            str(row["endpoint"]),
            str(row["count"]),
            f'{float(row["min"]):.2f}',
        ]
        for p in percentiles:
            values.append(f'{float(row[f"p{p:g}"]):.2f}')
        values.append(f'{float(row["max"]):.2f}')
        table_rows.append(values)
    widths = []
    for col in zip(headers, *table_rows):
        widths.append(max(len(str(cell)) for cell in col))
    def render(cells: Sequence[str]) -> str:
        return "  ".join(
            cell.ljust(width) if index == 0 else cell.rjust(width)
            for index, (cell, width) in enumerate(zip(cells, widths))
        )
    lines = [render(headers), "  ".join("-" * width for width in widths)]
    lines.extend(render(row) for row in table_rows)
    return "\n".join(lines)

```python
#!/usr/bin/env python3
"""Aggregate per-endpoint latency percentiles from JSON Lines logs."""

from __future__ import annotations

import argparse
import json
import math
import sys
from collections import defaultdict
from pathlib import Path
from typing import Iterable, Mapping, Sequence, TextIO

DEFAULT_PERCENTILES = (50.0, 90.0, 95.0, 99.0)


def parse_percentiles(text: str) -> tuple[float, ...]:
    """Parse a comma-separated list of percentile values."""
    parts = [part.strip() for part in text.split(",") if part.strip()]
    if not parts:
        raise argparse.ArgumentTypeError(
            "percentiles must contain at least one value"
        )

    values: list[float] = []
    for part in parts:
        try:
            value = float(part)
        except ValueError as exc:
            raise argparse.ArgumentTypeError(
                f"invalid percentile: {part!r}"
            ) from exc笔者

        if not math.isfinite(value) or not 0.0 <= value <= 100.0:
            raise argparse.ArgumentTypeError(
                f"percentile must be between 0 and 100: {part!r}"
            )

        values.append(value)

    return tuple(values)


def percentile(sorted_values: Sequence[float], p: float) -> float:
    """Return the p-th percentile using linear interpolation.

    ``sorted_values`` must already be sorted in ascending order.
    """
    if not sorted_values:
        raise ValueError("percentile() requires at least one value")
    if not math.isfinite(p) or not 0.0 <= p <= 100.0:
        raise ValueError("percentile must be between 0 and 100")

    if len(sorted_values) == 1:
        return float(sorted_values[0])

    rank = (len(sorted_values) - 1) * (p / 100.0)
    lower = math.floor(rank)
    upper = math.ceil(rank)

    if lower == upper:
        return float(sorted_values[lower])

    fraction = rank - lower
    lower_value = float(sorted_values[lower])
    upper_value = float(sorted_values[upper])
    return lower_value * (1.0 - fraction) + upper_value * fraction


def aggregate_json_lines(
    lines: Iterable[str],
    *,
    endpoint_field: str = "endpoint",
    latency_field: str = "latency_ms",
    source: str = "<stream>",
    warnings: TextIO | None = sys.stderr,
) -> dict[str, list[float]]:
    """Aggregate latency values by endpoint from JSON Lines input."""
    aggregates: dict[str, list[float]] = defaultdict(list)

    for line_no, raw_line in enumerate(lines, start=1):
        line = raw_line.strip()
        if not line:
            continue

        try:
            record = json.loads(line)
        except json.JSONDecodeError as exc:
            if warnings is not None:
                print(
                    f"warning: {source}:{line_no}: invalid JSON: {exc}",
                    file=warnings,
                )
            continue

        if not isinstance(record, dict):
            if warnings is not None:
                print(
                    f"warning: {source}:{line_no}: expected a JSON object",
                    file=warnings,
                )
            continue

        endpoint = record.get(endpoint_field)
        latency = record.get(latency_field)

        if endpoint is None:
            if warnings is not None:
                print(
                    f"warning: {source}:{line_no}: "
                    f"missing endpoint field {endpoint_field!r}",
                    file=warnings,
                )
            continue

        if latency is None:
            if warnings is not None:
                print(
                    f"warning: {source}:{line_no}: "
                    f"missing latency field {latency_field!r}",
                    file=warnings,
                )
            continue

        try:
            latency_value = float(latency)
        except (TypeError, ValueError):
            if warnings is not None:
                print(
                    f"warning: {source}:{line_no}: "
                    f"invalid latency {latency!r}",
                    file=warnings,
                )
            continue

        if not math.isfinite(latency_value):
            if warnings is not None:
                print(
                    f"warning: {source}:{line_no}: "
                    f"non-finite latency {latency!r}",
                    file=warnings,
                )
            continue

        aggregates[str(endpoint)].append(latency_value)

    return dict(aggregates)


def read_json_lines(
    path: str | Path,
    *,
    endpoint_field: str = "endpoint",
    latency_field: str = "latency_ms",
    warnings: TextIO | None = sys.stderr,
) -> dict[str, list[float]]:
    """Read a JSON Lines file and aggregate latency values by endpoint."""
    path = Path(path)
    with path.open("r", encoding="utf-8") as stream:
        return aggregate_json_lines(
            stream,
            endpoint_field=endpoint_field,
            latency_field=latency_field,
            source=str(path),
            warnings=warnings,
        )


def summarize(
    aggregates: Mapping[str, Sequence[float]],
    percentiles: Sequence[float] = DEFAULT_PERCENTILES,
) -> list[dict[str, object]]:
    """Build summary rows for each endpoint."""
    rows: list[dict[str, object]] = []

    for endpoint, values in aggregates.items():
        ordered = sorted(values)
        if not ordered:
            continue

        row: dict[str, object] = {
            "endpoint": endpoint,
            "count": len(ordered),
            "min": ordered[0],
            "max": ordered[-1],
        }

        for p in percentiles:
            row[f"p{p:g}"] = percentile(ordered, p)

        rows.append(row)

    return rows


def format_table(
    rows: Sequence[Mapping[str, object]],
    percentiles: Sequence[float],
) -> str:
    """Format summary rows as a plain-text table."""
    if not rows:
        return "No matching log records."

    percentile_labels = [f"P{p:g}" for p in percentiles]
    headers = ["Endpoint", "Count", "Min", *percentile_labels, "Max"]

    table_rows: list[list[str]] = []
    for row in rows:
        values = [
            str(row["endpoint"]),
            str(row["count"]),
            f"{float(row['min']):.2f}",
        ]

        for p in percentiles:
            values.append(f"{float(row[f'p{p:g}']):.2f}")

        values.append(f"{float(row['max']):.2f}")
        table_rows.append(values)

    widths: list[int] = []
    for column in zip(headers, *table_rows):
        widths.append(max(len(str(cell)) for cell in column))

    def render(cells: Sequence[str]) -> str:
        return "  ".join(
            cell.ljust(width) if index == 0 else cell.rjust(width)
            for index, (cell, width) in enumerate(zip(cells, widths))
        )

    lines = [
        render(headers),
        "  ".join("-" * width for width in widths),
    ]
    lines.extend(render(row) for row in table_rows)
    return "\n".join(lines)


def build_arg_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        description=(
            "Aggregate per-endpoint latency percentiles from a JSON Lines "
            "log file and print a table."
        )
    )
    parser.add_argument(
        "logfile",
        type=Path,
        help="Path to the JSON Lines log file.",
    )
    parser.add_argument(
        "--endpoint-field",
        default="endpoint",
        help="JSON field containing the endpoint name. Default: endpoint",
    )
    parser.add_argument(
        "--latency-field",
        default="latency_ms",
        help="JSON field containing the latency value. Default: latency_ms",
    )
    parser.add_argument(
        "--percentiles",
        type=parse_percentiles,
        default=DEFAULT_PERCENTILES,
        metavar="P1,P2,...",
        help=(
            "Comma-separated percentiles to compute. "
            "Default: 50,90,95,99"
        ),
    )
    parser.add_argument(
        "--sort",
        choices=("endpoint", "count"),
        default="endpoint",
        help="Sort table rows by endpoint or count. Default: endpoint",
    )
    return parser


def main(argv: Sequence[str] | None = None) -> int:
    parser = build_arg_parser()
    args = parser.parse_args(argv)

    aggregates = read_json_lines(
        args.logfile,
        endpoint_field=args.endpoint_field,
        latency_field=args.latency_field,
    )

    if not aggregates:
        print("No matching log records.", file=sys.stderr)
        return 1

    rows = summarize(aggregates, args.percentiles)

    if args.sort == "count":
        rows.sort(
            key=lambda row: (
                -int(row["count"]),
                str(row["endpoint"]),
            )
        )
    else:
        rows.sort(key=lambda row: str(row["endpoint"]))

    print(format_table(rows, args.percentiles))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
