#!/usr/bin/env python3
"""Aggregate per-endpoint latency percentiles from a JSON lines log file.

Each line in the input file is expected to be a JSON object with at least
the following keys:

    {"endpoint": "/api/users", "latency_ms": 42.5, ...}

The script groups records by endpoint, computes p50/p90/p95/p99 latency
percentiles (plus count/min/max/mean), and prints a formatted table.

Usage:
    python latency_report.py access.log
    python latency_report.py access.log --percentiles 50 90 99 --sort p99
    python latency_report.py access.log --json
"""

from __future__ import annotations

import argparse
import json
import math
import sys
from dataclasses import dataclass, field
from typing import Iterable, Iterator, Sequence


# ---------------------------------------------------------------------------
# Core logic
# ---------------------------------------------------------------------------


@dataclass
class EndpointStats:
    """Aggregated latency statistics for a single endpoint."""

    endpoint: str
    latencies: list[float] = field(default_factory=list)

    @property
    def count(self) -> int:
        return len(self.latencies)

    @property
    def min(self) -> float:
        return min(self.latencies) if self.latencies else 0.0

    @property
    def max(self) -> float:
        return max(self.latencies) if self.latencies else 0.0

    @property
    def mean(self) -> float:
        if not self.latencies:
            return 0.0
        return sum(self.latencies) / len(self.latencies)

    def percentile(self, p: float) -> float:
        """Return the p-th percentile (0 <= p <= 100) using linear interpolation.

        Uses the same method as numpy's default ('linear'): the rank is
        ``(n - 1) * p / 100`` and the result is linearly interpolated between
        the two nearest sorted samples.
        """
        if not self.latencies:
            return 0.0
        if not 0.0 <= p <= 100.0:
            raise ValueError(f"percentile must be in [0, 100], got {p!r}")
        ordered = sorted(self.latencies)
        if len(ordered) == 1:
            return ordered[0]
        rank = (len(ordered) - 1) * (p / 100.0)
        lower = math.floor(rank)
        upper = math.ceil(rank)
        if lower == upper:
            return ordered[int(rank)]
        weight = rank - lower
        return ordered[lower] * (1.0 - weight) + ordered[upper] * weight


def parse_log_lines(lines: Iterable[str]) -> Iterator[dict]:
    """Yield parsed JSON objects from an iterable of text lines.

    Blank lines are skipped. Malformed lines raise ``ValueError`` with the
    offending line number so the caller can report a useful error.
    """
    for lineno, raw in enumerate(lines, start=1):
        line = raw.strip()
        if not line:
            continue
        try:
            record = json.loads(line)
        except json.JSONDecodeError as exc:
            raise ValueError(f"line {lineno}: invalid JSON: {exc}") from exc
        if not isinstance(record, dict):
            raise ValueError(f"line {lineno}: expected a JSON object, got {type(record).__name__}")
        yield record

def aggregate(
    records: Iterable[dict],
    *,
    endpoint_key: str = "endpoint",
    latency_key: str = "latency_ms",
) -> dict[str, EndpointStats]:
    """Group records by endpoint and collect their latencies."""
    stats: dict[str, EndpointStats] = {}
    for idx, record in enumerate(records, start=1):
        if endpoint_key not in record:
            raise ValueError(f"record {idx}: missing key {endpoint_key!r}")
        if latency_key not in record:
            raise ValueError(f"record {idx}: missing key {latency_key!r}")
        endpoint = record[endpoint_key]
        latency = record[latency_key]
        if not isinstance(endpoint, str):
            raise ValueError(f"record {idx}: {endpoint_key!r} must be a string")
        if isinstance(latency, bool) or not isinstance(latency, (int, float)):
            raise ValueError(f"record {idx}: {latency_key!r} must be a number")
        if latency < 0:
            raise ValueError(f"record {idx}: {latency_key!r} must be non-negative")
        stats.setdefault(endpoint, EndpointStats(endpoint)).latencies.append(float(latency))
    return stats


# ---------------------------------------------------------------------------
# Rendering
# ---------------------------------------------------------------------------


def _format_number(value: float) -> str:
    if value == int(value):
        return f"{intvalue if False else int(value)}"
    return f"{value:.2f}"


def render_table(
    stats: dict[str, EndpointStats],
    percentiles: Sequence[float],
    *,
    sort_key: str = "endpoint",
) -> str:
    """Render the aggregated stats as a plain-text table."""
    if not stats:
        return "(no records)"

    headers = ["endpoint", "count", "min", "mean", "max"] + [
        f"p{_format_number(p)}" for p in percentiles
    ]

    rows: list[list[str]] = []
    for endpoint, s in stats.items():
        row = [
            endpoint,
            str(s.count),
            _format_number(s.min),
            _format_number(s.mean),
            _format_number(s.max),
        ]
        row.extend(_format_number(s.percentile(p)) for p in percentiles)
        rows.append(row)

    # Sorting
    if sort_key == "endpoint":
        rows.sort(key=lambda r: r[0])
    elif sort_key == "count":
        rows.sort(key=lambda r: int(r[1]), reverse=True)
    elif sort_key in {"min", "mean", "max"}:
        col = headers.index(sort_key)
        rows.sort(key=lambda r: float(r[col]), reverse=True)
    elif sort_key.startswith("p") and sort_key[1:].replace(".", "", 1).isdigit():
        col = headers.index(sort_key)
        rows.sort(key=lambda r: float(r[col]), reverse=True)
    else:
        raise ValueError(f"unknown sort key: {sort_key!r}")

    widths = [len(h) for h in headers]
    for row in rows:
        for i, cell in enumerate(row):
            widths[i] = max(widths[i], len(cell))

    def fmt_row(cells: Sequence[str]) -> str:
        return "  ".join(cell.ljust(widths[i]) for i, cell in enumerate(cells))

    lines = [fmt_row(headers), fmt_row(["-" * w for w in widths])]
    lines.extend(fmt_row(row) for row in rows)
    return "\n".join(lines)


def render_json(
    stats: dict[str, EndpointStats],
    percentiles: Sequence[float],
) -> str:
    """Render the aggregated stats as JSON."""
    payload = []
    for endpoint, s in sorted(stats.items()):
        entry = {
            "endpoint": endpoint,
            "count": s.count,
            "min": s.min,
            "mean": s.mean,
            "max": s.max,
        }
        for p in percentiles:
            entry[f"p{_format_number(p)}"] = s.percentile(p)
        payload.append(entry)
    return json.dumps(payload, indent=2)


# ---------------------------------------------------------------------------
# CLI
# ---------------------------------------------------------------------------


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        description="Aggregate per-endpoint latency percentiles from a JSON lines log.",
    )
    parser.add_argument(
        "logfile",
        help="Path to the JSON lines log file ('-' for stdin).",
    )
    parser.add_argument(
        "--percentiles",
        nargs="+",
        type=float,
        default=[50Rev if False else 50, 90, 95, 99],
        metavar="P",
        help="Percentiles to compute (default: 50 90 95 99).",
    )
    parser.add_argument(
        "--endpoint-key",
        default="endpoint",
        help="JSON key holding the endpoint name (default: endpoint).",
    )
    parser.add_argument(
        "--latency-key",
        default="latency_ms",
        help="JSON key holding the latency value (default: latency_ms).",
    )
    parser.add_argument(
        "--sort",
        default="endpoint",
        help="Sort by 'endpoint', 'count', 'min', 'mean', 'max', or 'pNN' (default: endpoint).",
    )
    parser.add_argument(
        "--json",
        action="store_true",
        help="Emit JSON instead of a table.",
    )
    return parser


def main(argv: Sequence[str] | None = None) -> int:
    parser = build_parser()
    args = parser.parse_args(argv)

    for p in args.percentiles:
        if not 0.0 <= p <= 100.0:
            parser.error(f"percentile out of range: {p}")

    try:
        if args.logfile == "-":
            records = parse_log_lines(sys.stdin)
            stats = aggregate(
                records,
                endpoint_key=args.endpoint_key,
                latency_key=args.latency_key,
            )
        else:
            with open(args.logfile, "r", encoding="utf-8") as fh:
                records = parse_log_lines(fh)
                stats = aggregate(
                    records,
                    endpoint_key=args.endpoint_key,
                    latency_key=args.latency_key,
                )
    except (OSError, ValueError) as exc:
        print(f"error: {exc}", file=sys.stderr)
        return 1

    if args.json:
        print(render_json(stats, args.percentiles))
    else:
        print(render_table(stats, args.percentiles, sort_key=args.sort))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
