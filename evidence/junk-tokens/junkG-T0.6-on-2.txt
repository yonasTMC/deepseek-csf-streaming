```python
# healthcheck.py
"""Async HTTP health checker with retries, exponential backoff, and reporting."""
from __future__ import annotations

import argparse
import asyncio
import sys
import time
from dataclasses import dataclass, field
from enum import Enum
from typing import Iterable, Sequence

import aiohttp


class Status(str, Enum):
    HEALTHY = "healthy"
    UNHEALTHY = "unhealthy"


@dataclass
class CheckResult:
    url: str
    status: Status
    attempts: int
    latency_ms: float
    status_code: int | None = None
    error: str | None = None
    history_ms: list[float] = field(default_factory=list)


class HealthChecker:
    def __init__(
        self,
        urls: Sequence[str],
        *,
        timeout: float = 5.0,
        retries: int = 3,
        base_backoff: float = 0.5,
        max_backoff: float = 10.0,
        expected_status: int = 200,
        concurrency: int = 10,
        session: aiohttp.ClientSession | None = None,
    ) -> None:
        if retries < 1:
            raise ValueError("retries must be >= 1")
        if concurrency < 1:
            raise ValueError("concurrency must be >= 1")
        self.urls = list(urls)
        self.timeout = timeout
        self.retries = retries
        self.base_backoff = base_backoff
        self.max_backoff = max_backoff
        self.expected_status = expected_status
        self.concurrency = concurrency
        self._session = session

    def _backoff(self, attempt: int) -> float:
        # attempt is 1-based; delay before the next try
        return min(self.base_backoff * (2 ** (attempt - 1)), self.max_backoff)

    async def _check_one(self, session: aiohttp.ClientSession, url: str) -> CheckResult:
        history: list[float] = []
        last_error: str | None = None
        last_status: int | None = None
        timeout = aiohttp.ClientTimeout(total=self.timeout)

        for attempt in range(1, self.retries + 1):
            start = time.perf_counter()
            try:
                async with session.get(url, timeout=timeout) as resp:
                    last_status = resp.status
                    await resp.read()
                    latency = (time.perf_counter() - start) * 1000.0
                    history.append(latency)
                    if resp.status == self.expected_status:
                        return CheckResult(
                            url=url,
                            status=Status.HEALTHY,
                            attempts=attempt,
                            latency_ms=latency,
                            status_code=resp.status,
                            history_ms=history,
                        )
                    last_error = f"unexpected status {resp.status}"
            except (aiohttp.ClientError, asyncio.TimeoutError) as exc:
                latency = (time.perf_counter() - start) * 1000.0
                history.append(latency)
                last_error = f"{type(exc).__name__}: {exc}"

            if attempt < self.retries:
                await asyncio.sleep(self._backoff(attempt))

        return CheckResult(
            url=url,
            status=Status.UNHEALTHY,
            attempts=self.retries,
            latency_ms=history[-1] if history else 0.0,
            status_code=last_status,
            error=last_error,
            history_ms=history,
        )

    async def _run_with(self, session: aiohttp.ClientSession) -> list[CheckResult]:
        sem = asyncio.Semaphore(self.concurrency)

        async def guarded(url: str) -> CheckResult:
            async with sem:
                return await self._check_one(session, url)

        return await asyncio.gather(*(guarded(u) for u in self.urls))

    async def run(self) -> list[CheckResult]:
        if self._session is not None:
            return await self._run_with(self._session)
        async with aiohttp.ClientSession() as session:
            return await self._run_with(session)


def format_report(results: Iterable[CheckResult]) -> str:
    results = list(results)
    total = len(results)
    healthy = sum(1 for r in results if r.status is Status.HEALTHY)
    unhealthy = total - healthy
    lines = [
        "Health Check Report",
        "===================",
        f"Total: {total}  Healthy: {healthy}  Unhealthy: {unhealthy}",
        "",
    ]
    for r in results:
        if r.status is Status.HEALTHY:
            lines.append(
                f"[ OK ] {r.url}  status={r.status_code}  "
                f"latency={r.latency_ms:.1f}ms  attempts={r.attempts}"
            )
        else:
            lines.append(
                f"[FAIL] {r.url}  status={r.status_code or '-'}  "
                f"latency={r.latency_ms:.1f}ms  attempts={r.attempts}  "
                f"error={r.error}"
            )
    return "\n".join(lines)


def main(argv: Sequence[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="Async HTTP health checker")
    parser.add_argument("urls", nargs="+", help="URLs to check")
    parser.add_argument("--timeout", type=float, default=5.0)
    parser.add_argument("--retries", type=int, default=3)
    parser.add_argument("--base-backoff", type=float, default=0.5)
    parser.add_argument("--max-backoff", type=float, default=10.0)
    parser.add_argument("--concurrency", type=int, default=10)
    parser.add_argument("--expected-status", type=int, default=200)
    args = parser.parse_args(argv)

    checker = HealthChecker(
        args.urls,
        timeout=args.timeout,
        retries=args.retries,
        base_backoff=args.base_backoff,
        max_backoff=args.max_backoff,
        concurrency=args.concurrency,
        expected_status=args.expected_status,
    )
    results = asyncio.run(checker.run())
    print(format_report(results))
    return 0 if all(r.status is Status.HEALTHY for r in results) else 1


if __name__ == "__main__":
    sys.exit(main())
```
