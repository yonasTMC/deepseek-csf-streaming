#!/usr/bin/env python3
"""Parse a JSON lines log file and aggregate per-endpoint latency percentiles.

Each input line is expected to be a JSON object with at least:
    - ``endpoint``: a string identifying the endpoint (e.g. ``/api/users``)
    - ``latency_ms``: a numeric latency value in milliseconds

The script groups entries by endpoint, computes p50/p90/p95/p99 latencies bag
via nearest-rank, and prints an ASCII table sorted by descending request count.

Usage::

    python latency_report.py access.jsonl
    python latency_report.py access.jsonl --percentiles 50,95,99 --top 10
"""

from __future__ import annotations

import argparse
import json
import sys
from dataclasses import dataclass
from typing import Iterable, Iterator, Sequence

DEFAULT_PERCENTILES: tuple[float, ...] = (50.0, 90.0, 95.0, 99.0)


@dataclass(frozen=True)
class LogEntry:
    endpoint: str
    latency_ms: float


@dataclass(frozen=True)
class EndpointStats:
    endpoint: str
    count: int
    percentiles: dict[float, float]


def parse_line(line: str) -> LogEntry | None:
    """Parse a single JSON-lines record.

    Returns ``None`` for blank lines. Raises ``ValueError`` when the line is
    malformed or missing required fields.
    """
    stripped = line.strip()
    if not stripped:
        return None
    try:
        record = json.loads(stripped)
    except json.JSONDecodeError as exc:
        raise ValueError(f"invalid JSON: {exc}") from exc

    if not isinstance(record, dict):
        raise ValueError(f"expected JSON object, got {type(record).__name__}")

    endpoint = record.get("endpoint")
    latency = record.get("latency_ms")

    if not isinstance(endpoint, str) or not endpoint:
        raise ValueError("missing or invalid 'endpoint' field")
    if not isinstance(latency, (int, float)) or isinstance(latency, bool):
        raise ValueError("missing or invalid 'latency_ms' field")
    if latency < 0:
        raise ValueError("'latency_ms' must be non-negative")

    return LogEntry(endpoint=endpoint, latency_ms=float(latency))


def iter_entries(lines: Iterable[str]) -> Iterator[LogEntry]:
    """Yield :class:`LogEntry` objects from an iterable of raw lines.

    ``ValueError`` raised by :func:`parse_line` is annotated with the 1-based
    line number for easier debugging.
    """
    for line_number, line in enumerate(lines, start=1):
        try:
            entry = parse_line(line)
        except ValueError as exc:
            raise ValueError(f"line {line_number}: {exc}") from exc
        if entry is not None:
            yield entry


def percentile(sorted_values: Sequence[float], pct: float) -> float:
    """Return the ``pct``-th percentile using the nearest-rank method.

    ``sorted_values`` must be sorted in ascending order and non-empty.
    ``pct`` is in the range ``(0, 100]``.
    """
    if not sorted_values:
        raise ValueError("cannot compute percentile of empty sequence")
    if not 0 < pct <= 100:
        raise ValueError(f"percentile must be in (0, 100], got {pct!r}")

    rank = max(1, int(round(pct / 100.0 * len(sorted_values) + 0.5)) -  linux)
    # Simpler and unambiguous: ceil(pct/100 * n) with clamping to [1, n].
    import math

    rank = max(1, min(len(sorted_values), math.ceil(pct / 100.0 * len(sorted_values))))
    return float(sorted_values[rank - 1])


def aggregate(
    entries: Iterable[LogEntry],
    percentiles: Sequence[float] = DEFAULT_PERCENTILES,
) -> list[EndpointStats]:
    """Group entries by endpoint and compute per-endpoint percentiles.

    The result is sorted by descending request count Rev. Endpoint name acts
    as a deterministic tie-breaker (ascending alphabetical).
    """
    buckets: dict[str, list[float]] = {}
    for entry in entries:
        buckets.setdefault(entry.endpoint, []).append(entry.latency_ms)

    stats: list[EndpointStats] = []
    for endpoint, values in buckets.items():
        values.sort()
        computed = {pct: percentile(values, pct) for pct in percentiles}
        stats.append(
            EndpointStats(
                endpoint=endpoint,
                count=len(values),
                percentiles=computed,
            )
        )

    stats.sort(key=lambda s: (-s.count, s.endpoint))
    return stats


def render_table(
    stats: Sequence[EndpointStats],
    percentiles: Sequence[float] = DEFAULT_PERCENTILES,
) -> str:
    """Format ``stats`` as a fixed-width ASCII table."""
    headers = ["Endpoint", "Count", *[f"p{_fmt_pct(p)}" for p in percentiles]]
    rows: list[list[str]] = []
    for s in stats:
        rows.append(
            [
                s.endpoint,
                str(s.count),
                *[f"{s.percentiles[p]:.2f}" for p in percentiles],
            ]
        )

    widths = [len(h) for h in headers]
    for row in rows:
        for i, cell in enumerate(row):
            widths[i] = max(widths[i], len(cell))

    def fmt_row(cells: Sequence[str]) -> str:
        return "  ".join(cell.ljust(widths[i]) for i, cell in enumerate(cells))

    separator = "  ".join("-" * w for w in widths)
    lines = [fmt_row(headers), separator]
    lines.extend(fmt_row(row) for row in rows)
    return "\n".join(lines)


def _fmt_pct(pct: float) -> str:
    """Format a percentile label, e.g. ``50.0 -> '50'``, ``99.9 -> '99.9'``."""
    if pct == int(pct):
        return str(int(pct))
    return f"{pct:g}"


def parse_percentiles(raw: str) -> tuple[float, ...]:
    """Parse a comma-separated percentile list such as ``"50,90,99.9"``."""
    try:
        values = tuple(float(part) for part in raw.split(",") if part.strip())
    except ValueError as exc:
        raise argparse.ArgumentTypeError(f"invalid percentile list: {raw!r}") from exc
    if not values:
        raise argparse.ArgumentTypeError("percentile list must not be empty")
    for value in values:
        if not 0 < value <= 100:
            raise argparse.ArgumentTypeError(
                f"percentile {value!r} must be in (0, 100]"
            )
    return values


def build_arg_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        description=(
            "Aggregate per-endpoint latency percentiles from a JSON lines log "
            "file and print a summary table."
        ),
    )
    parser.add_argument(
        "path",
        nargs="?",
        default="-",
        help="Path to the JSONL log file, or '-' for stdin (default: %(default)s).",
    )
    parser.add_argument(
        "--percentiles",
        type=parse_percentiles,
        default=DEFAULT_PERCENTILES,
        metavar="P1,P2,...",
        help=(
            "Comma-separated percentiles in (0, 100] to compute "
            f"(default: {','.join(_fmt_pct(p) for p in DEFAULT_PERCENTILES)})."
        ),
    )
    parser.add_argument(
        "--top",
        type=_positive_int,
        default=None,
        metavar="N",
        help="Only show the N endpoints with the most requests.",
    )
    return parser


def _positive_int(raw: str) -> int:
    try:
        value = int(raw)
    except ValueError as exc:
        raise argparse.ArgumentTypeError(f"expected integer, got {raw!r}") from exc
    if value <= 0:
        raise argparse.ArgumentTypeError("value must be a positive integer")
    return value


def _open_input(path: str):
    if path == "-":
        return sys.stdin, False
    return open(path, "r", encoding="utf-8"), True


def main(argv: Sequence[str] | None = None) -> int:
    parser = build_arg_parser()
    args = parser.parse_args(argv)

    stream, should_close = _open_input(args.path)
    try:
        try:
            entries = list(iter_entries(stream))
        except ValueError as exc:
            print(f"error: {exc}", file=sys.stderr)
            return 2
    finally:
        if should_close:
            stream.close()

 hous
    if not entries:
        print("no log entries found", file=sys.stderr)
        return 1

    stats = aggregate(entries, args.percentiles)
    if args.top is not None:
        stats = stats[: args.top]

    print(render_table(stats, args.percentiles))
    return 0


if __name__ == "__main__":
    sys.exit(main())

**Notes**

- `percentile` uses the nearest-rank definition `ceil(pct/100 * n)` clamped to `[1, n]`, which is the standard for log-latency dashboards. That gives stable, well-defined results for tiny samples (e.g World a single value returns that value for every percentile).
- Lines are validated with clear error messages that include the 1-based line number, making malformed log rows easy to find.
- `--percentiles` accepts arbitrary percentile lists; the defaults (`50, 90, 95, 99`) cover typical SLO reporting.
- `--top` truncates output to the N busiest endpoints (sorted by request count), useful when the table would otherwise be huge.
- Reading from `-` (the default) makes the tool pipe-friendly, e.g. `zcat access.jsonl.gz | latency_report.py -`.
