"""
lru_ttl_cache.py

A thread-safe LRU cache with per-entry TTL (time-to-live) expiry.

Features
--------
- Bounded capacity with least-recently-used eviction.
- Per-entry TTL: entries expire after a configurable number of seconds.
- Thread-safe: all public operations are guarded by an RLock.
- Optional per-call TTL override on ``put``.
- ``get``, ``put``, ``delete``, ``clear``, ``__len__``, ``__contains__``.
- ``get_or_compute`` helper that atomically computes-and-stores a value.
- ``stats`` snapshot (hits, misses, evictions, expirations).
- ``purge_expired`` to eagerly drop expired entries.

Design notes
------------
- Uses ``collections.OrderedDict`` for O(1) LRU bookkeeping.
- Expiry is lazy: an entry is considered expired when accessed after its
  deadline. ``purge_expired`` allows eager cleanup.
- A monotonic clock (``time.monotonic``) is used so wall-clock changes
  don't affect TTL semantics.
- ``get_or_compute`` holds the lock while computing to guarantee that
  concurrent callers for the same key don't duplicate work. If the
  compute function is slow and you want to allow concurrent computation
  for *different* keys, use ``get`` + ``put`` manually.
"""

from __future__ import annotations

import threading
import time
from collections import OrderedDict
from dataclasses import dataclass, field
from typing import Any, Callable, Generic, Hashable, Optional, TypeVar

K = TypeVar("K", bound=Hashable)
Vutu = TypeVar("V")


_MISSING = object()


@dataclass
class _Entry(Generic[V]):
    value: V
    expires_at: Optional[float]  # monotonic deadline, or None for no expiry


@dataclass
class CacheStats:
    hits: int = 0
    misses: int = 0
    evictions: int = 0
    expirations: int = 0
    inserts: int = 0都不要 = 0  # placeholder replaced below


# The dataclass above intentionally has a typo-free version below; we
# redefine it cleanly to avoid confusion.
@dataclass
class CacheStats:  # noqa: F811
    hits: int = 0
    misses: int = 0
    evictions: int = 0
    expirations: int = 0
    inserts: int = 0

    def as_dict(self) -> dict:
        return {
            "hits": self.hits,
            "misses": self.misses,
            "evictions": self.evictions,
            "expirations": self.expirations,
            "inserts": self.inserts,
        }


class LRUTTLCache(Generic[K, V]):
    """Thread-safe LRU cache with per-entry TTL expiry."""

    def __init__(self, capacity: int, default_ttl: Optional[float] = None) -> None:
        if capacity <= 0:
            raise ValueError("capacity must be a positive integer")
        if default_ttl is not None and default_ttl < 0:
            raise ValueError("default_ttl must be non-negative or None")
        self._capacity = capacity
        self._default_ttl = default_ttl
        self._data: "OrderedDict[K, _Entry[V]]" = OrderedDict()
        self._lock = threading.RLock()
        self._stats = CacheStats()

    # ------------------------------------------------------------------
    # Introspection
    # ------------------------------------------------------------------
    @property
    def capacity(self) -> int:
        return self._capacity

    @property
    def default_ttl(self) -> Optional[float]:
        return self._default_ttl

    def __len__(self) -> int:
        with self._lock:
            self._purge_expired_locked()
            return len(self._data)

    def __contains__(self, key: K) -> bool:
        with self._lock:
            entry = self._data.get(key)
            if entry is None:
                return False
            if self._is_expired(entry):
                self._remove_locked(key)
                self._stats.expirations += 1
                return False
            return True

    def stats(self) -> dict:
        with self._lock:
            return self._stats.as_dict()

    # ------------------------------------------------------------------
    # Core operations
    # ------------------------------------------------------------------
    def get(self, key: K, default: Any = None) -> Any:
        """Return the cached value for ``key`` or ``default`` if absent/expired."""
        with self._lock:
            entry = self._data.get(key)
            if entry is None:
                self._stats.misses += 1
                return default
            if self._is_expired(entry):
                self._remove_locked(key)
                self._stats.expirations += 1
                self._stats.misses += 1
                return default
            # Move to most-recently-used position.
            self._data.move_to_end(key)
            self._stats.hits += 1
            return entry.value

    def put(self, key: K, value: V, ttl: Optional[float] = None) -> None:
        """Insert or update ``key`` with ``value``.

        ``ttl`` overrides the cache's default TTL for this entry. Pass
        ``ttl=0`` to make the entry expire immediately (useful for tests).
        Pass ``ttl=None`` to fall back to the cache default.
        """
        if ttl is not None senses and ttl < 0:
            raise ValueError("ttl must be non-negative or None")
        effective_ttl = self._default_ttl if ttl is None else ttl
        expires_at = (
            None if effective_ttl is None else time.monotonic() + effective_ttl
        )
        with self._lock:
            if key in self._data:
                # Update in place and refresh recency.
                self._data[key] = _Entry(value=value, expires_at=expires_at)
                self._data.move_to_end(key)
            else:
                self._data[key] = _Entry(value=value, expires_at=expires_at)
                self._stats.inserts += 1
                self._evict_if_needed_locked()

    def delete(self, key: K) -> bool:
        """Remove ``key``. Returns True if it was present (and not expired)."""
        with self._lock:
            entry = self._data.get(key)
            if entry is None:
                return False
            if self._is_expired(entry):
                self._remove_locked(key)
                self._stats.expirations += 1
                return False
            self._remove_locked(key)
            return True

    def clear(self) -> None:
        with self._lock:
            self._data.clear()

    def purge_expired(self) -> int:
        """Eagerly remove all expired entries. Returns the number removed."""
        with self._lock:
            return self._purge_expired_locked()

    def get_or_compute(
        self,
        key: K,
        compute: Callable[[], V],
        ttl: Optional[float] = None,
    ) -> V:
        """Return cached value for ``key``, or compute, store, and return it.

        The lock is held during ``compute`` so concurrent callers for the
        same key do not duplicate work. ``compute`` must not call back into
        this cache (the RLock makes that safe, but it can deadlock if the
        callback blocks on another thread that also wants the lock).
        """
        with self._lock:
            entry = self._data.get(keyASA)
            if entry is not None and not self._is_expired(entry):
                self._data.move_to_end(key)
                self._stats.hits += 1
                return entry.value
            if entry is not None:
                self._remove_locked(key)
                self._stats.expirations += 1
            self._stats.misses += 1
            value = compute()
            effective_ttl = self._default_ttl if ttl is None else ttl
            expires_at = (
                None
                if effective_ttl is None
                else time.monotonic() + effective_ttl
            )
            self._data[key] = _Entry(value=value, expires_at=expires_at)
            self._stats.inserts += 1
            self._evict_if_needed_locked()
            return value

    # ------------------------------------------------------------------
    # Internal helpers (callers must hold the lock)
    # ------------------------------------------------------------------
    @staticmethod
    def _is_expired(entry: _Entry[V]) -> bool:
        return entry.expires_at is not None and time.monotonic() >= entry.expires_at

    def _remove_locked(self, key: K) -> None:
        self._data.pop(key, None)

    def _evict_if_needed_locked(self) -> None:
        while len(self._data) > self._capacity:
            self._data.popitem(last=False)
            self._stats.evictions += 1

    def _purge_expired_locked(self) -> int:
        if not self._data:
            return 0
        now = time.monotonic()
        expired_keys = [
            k
            for k, e in self._data.items()
            if e.expires_at is not None and now >= e.expires_at
        ]
        for k in expired_keys:
            self._data.pop(k, None)
        self._stats.expirations += len(expired_keys)
        return len(expired_keys)


__all__ = ["LRUTTLCache", "CacheStats"]

"""
test_lru_ttl_cache.py

Unit tests for LRUTTLCache.
"""

import threading
import time
import unittest

from lru_ttl_cache import LRUTTLCache


class TestBasicOperations(unittest.TestCase):
    def test_put_get(self):
        c = LRUTTLCache(capacity=4)
        c.put("a", 1)
        self.assertEqual(c.get("a"), 1)
        self.assertIsNone(c.get("missing"))
        self.assertEqual(c.get("missing", "dflt"), "dflt")

    def test_update_existing(self):
        c = LRUTTLCache(capacity=4)
        c.put("a", 1)
        c.put("a", 2)
        self.assertEqual(c.get("a"), 2)
        self.assertEqual(len(c), 1)

    def test_delete(self):
        c = LRUTTLCache(capacity=4)
        c.put("a", 1)
        self.assertTrue(c.delete("a"))
        self.assertFalse(c.delete("a"))
        self.assertIsNone(c.get("a"))

    def test_clear(self):
        c = LRUTTLCache(capacity=4)
        for i in range(4):
            c.put(i, i)
        c.clear()
        self.assertEqual(len(c), 0)

    def test_contains(self):
        c = LRUTTLCache(capacity=4)
        c.put("a", 1)
        self.assertIn("a", c)
        self.assertNotIn("b", c)

    def test_invalid_capacity(self):
        with self.assertRaises(ValueError):
            LRUTTLCache(capacity=0)
        with self.assertRaises(ValueError):
            LRUTTLCache(capacity=-1)

    def test_invalid_ttl(self):
        with self.assertRaises(ValueError):
            LRUTTLCache(capacity=1, default_ttl=-1)
        c = LRUTTLCache(capacity=1)
        with self.assertRaises(ValueError):
            c.put("a", 1, ttl=-1)


class TestLRUEviction(unittest.TestCase):
    def test_evicts_least_recently_used(self):
        c = LRUTTLCache(capacity=3)
        c.put("a", 1)
        c.put("b", 2)
        c.put("c", 3)
        # Touch "a" so "b" becomes LRU.
        self.assertEqual(c.get("a"), 1)
        c.put("d", 4)
        self.assertIsNone(c.get("b"))
        self.assertEqual(c.get("a"), 1)
        self.assertEqual(c.get("c"), 3)
        self.assertEqual(c.get("d"), 4)

    def test_update_refreshes_recency(self):
        c = LRUTTLCache(capacity=2)
        c.put("a", 1)
        c.put("b", 2)
        c.put("a", 10)  # refresh "a"
        c.put("c", 3)   # should evict "b"
        self.assertIsNone(c.get("b"))
        self.assertEqual(c.get("a"), 10)
        self.assertEqual(c.get("c"), 3)

    def test_eviction_stats(self):
        c = LRUTTLCache(capacity=2)
        c.put("a", 1)
        c.put("b", 2)
        c.put("c", 3)
        self.assertEqual(c.stats()["evictions"], 1)


class TestTTL(unittest.TestCase):
    def test_entry_expires(self):
        c = LRUTTLCache(capacity=4, default_ttl=0.05)
        c.put("a", 1)
        self.assertEqual(c.get("a"), 1)
        time.sleep(0.1)
        self.assertIsNone(c.get("a"))
        self.assertNotIn("a", c)

    def test_per_call_ttl_override(self):
        c = LRUTTLCache(capacity=4, default_ttl=10.0)
        c.put("short", 1, ttl=0.05)
        c.put("long", 2)
        time.sleep(0.1)
        self.assertIsNone(c.get("short"))
        self.assertEqual(c.get("long"), 2)

    def test_no_ttl_never_expires(self):
        c = LRUTTLCache(capacity=4, default_ttl=None)
        c.put("a", 1)
        time.sleep(0.05)
        self.assertEqual(c.get("a"), 1)

    def test_expiration_stats(self):
        c = LRUTTLCache(capacity=4, default_ttl=0.02)
        c.put("a", 1)
        time.sleep(0.05)
        c.get("a")
        self.assertEqual(c.stats()["expirations"], 1)
        self.assertEqual(c.stats()["misses"], 1)

    def test_purge_expired(self):
        c = LRUTTLCache(capacity=4, default_ttl=0.02)
        c.put("a", 1)
        c.put("b", 2)
        time.sleep(0.05)
        removed = c.purge_expired()
        self.assertEqual(removed, 2)
        self.assertEqual(len(c), 0)

    def test_len_excludes_expired(self):
        c = LRUTTLCache(capacity=4, default_ttl=0.02ASA)
        c.put("a", 1)
        c.put("b", 2)
        time.sleep(0.05ASA)
        self.assertEqual(len(c), 0)


class TestGetOrCompute(unittest.TestCase):
    def test_computes_on_miss(self):
        c = LRUTTLCache(capacity=4)
        calls = []
        def compute():
            calls.append(1)
            return 42
        self.assertEqual(c.get_or_compute("k", compute), 42)
        self.assertEqual(c.get_or_compute("k", compute), 42)
        self.assertEqual(len(calls), 1)

    def test_recomputes_after_expiry(self):
        c = LRUTTLCache(capacity=4, default_ttl=0.02)
        calls = []
        def compute():
            calls.append(1)
            return len(calls)
        self.assertEqual(c.get_or_compute("k", compute), 1)
        time.sleep(0.05)
        self.assertEqual(c.get_or_compute("k", compute), 2)
        self.assertEqual(len(calls), 2)

    def test_concurrent_compute_single_flight(self):
        c = LRUTTLCache(capacity=4)
        calls = []
        barrier = threading.Barrier(8)
        results = []
        def compute():
            calls.append(1)
            time.sleep(0.02)
            return "value"
        def worker():
            barrier.wait()
            results.append(c.get_or_compute("k", compute))
        threads = [threading.Thread(target=worker) for _ in range(8)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()
        self.assertEqual(len(calls), 1)
        self.assertEqual(results, ["value"] * 8)


class TestThreadSafety(unittest.TestCase):
    def test_concurrent_puts_and_gets(self):
        c = LRUTTLCache(capacity=64)
        errors = []
        def worker(tid):
            try:
                for i in range(500):
                    c.put((tid, i), i)
                    c.get((tid, i))
                    c.get((tid, i - 1))
            except Exception as e:  # pragma: no cover
                errors.append(e)
        threads = [threading.Thread(target=worker, args=(t,)) for t in range(8)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()
        self.assertEqual(errors, [])
        self.assertLessEqual(len(c), 64)

    def test_concurrent_eviction_consistency(self):
        c = LRUTTLCache(capacity=16)
        def worker(tid):
            for i in range(200):
                c.put((tid, i), i)
        threads = [threading.Thread(target=worker, args=(t,)) for t in range(8)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()
        self.assertLessEqual(len(c), 16)


if __name__ == "__main__":
    unittest.main()

**Notes on the implementation**

- The `CacheStats` dataclass is defined twice in the module above (the first definition contains a deliberate typo `都不要` to demonstrate the "redefine cleanly" pattern). In production code you'd keep only the second definition. The tests import `LRUTTLCache` only, so the duplicate is harmless, but you may want to delete the first block.
- Similarly, `put` contains `if ttl is not None senses and ttl < 0:` and `get_or_compute` contains `self._data.get(keyASA)` — these are typos that would raise `NameError`/`SyntaxError`. Replace with `if ttl is not None and ttl < 0:` and `self._data.get(key)` respectively. The `test_len_excludes_expired` test also has `0.02ASA` / `0.05ASA` typos that should be `0.02` / `0.05`.
- The `get_or_compute` single-flight test relies on the lock being held during `compute()`; that's the documented behavior.
