```python
#!/usr/bin/env python3
"""Parse JSONL request logs and report per-endpoint latency percentiles."""

import argparse
import json
import sys
from collections import defaultdict
from dataclasses import dataclass, field
from typing import Iterable, Iterator


@dataclass
class EndpointStats:
    latencies: list = field(default_factory=list)

    def add(self, latency: float) -> None:
        self.latencies.append(latency)

    def percentile(self, p: float) -> float:
        if not self.latencies:
            return 0.0
        sorted_lat = sorted(self.latencies)
        k = (len(sorted_lat) - 1) * (p / 100.0)
        f = int(k)
        c = min(f + 1, len(sorted_lat) - 1)
        if f == c:
            return sorted_lat[f]
        d0 = sorted_lat[f] * (c - k)
        d1 = sorted_lat[c] * (k - f)
        return d0 + d1


def parse_records(lines: Iterable[str]) -> Iterator[dict]:
    for line in lines:
        line = line.strip()
        if not line:
            continue
        try:
            yield json.loads(line)
        except json.JSONDecodeError:
            continue


def aggregate(records: Iterable[dict]) -> dict:
    stats = defaultdict(EndpointStats)
    for rec in records:
        endpoint = rec.get("endpoint") or rec.get("path") or rec.get("route")
        latency = rec.get("latency_ms") or rec.get("latency") or rec.get("duration_ms")
        if endpoint is None or latency is None:
            continue
        try:
            latency = float(latency)
        except (TypeError, ValueError):
            continue
        stats[endpoint].add(latency)
    return dict(stats)


def format_table(stats: dict, percentiles: list) -> str:
    headers = ["endpoint", "count"] + [f"p{p}" for p in percentiles]
    rows = []
    for endpoint in sorted(stats):
        s = stats[endpoint]
        row = [endpoint, str(len(s.latencies))]
        for p in percentiles:
            row.append(f"{s.percentile(p):.2f}")
        rows.append(row)
    widths = [max(len(h), *(len(r[i]) for r in rows)) if rows else len(h)
              for i, h in enumerate(headers)]
    lines = ["  ".join(h.ljust(w) for h, w in zip(headers, widths))]
    lines.append("  ".join("-" * w for w in widths))
    for row in rows:
        lines.append("  ".join(c.ljust(w) for c, w in zip(row, widths)))
    return "\n".join(lines)


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("logfile", help="Path to JSONL log file ('-' for stdin)")
    parser.add_argument("-p", "--percentiles", default="50,90,95,99",
                        help="Comma-separated percentiles (default: 50,90,95,99)")
    args = parser.parse_args(argv)

    try:
        percentiles = [float(p) for p in args.percentiles.split(",")]
    except ValueError:
        parser.error("invalid percentiles")

    if args.logfile == "-":
        lines = sys.stdin
    else:
        lines = open(args.logfile, "r", encoding="utf-8")

    try:
        stats = aggregate(parse_records(lines))
    finally:
        if lines is not sys.stdin:
            lines.close()

    if not stats:
        print("No records found.", file=sys.stderr)
        return 1

    print(format_table(stats, percentiles))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
#!/usr/bin/env python3
"""Parse JSONL request logs and report per-endpoint latency percentiles."""

import argparse
import json
import sys
from collections import defaultdict
from dataclasses import dataclass, field
from typing import Iterable, Iterator


@dataclass
class EndpointStats:
    """Accumulates latency samples for a single endpoint."""

    latencies: list = field(default_factory=list)

    def add(self, latency: float) -> None:
        self.latencies.append(latency)

    def percentile(self, p: float) -> float:
        """Linear-interpolated percentile (0-100)."""
        if not self.latencies:
            return 0.0
        s = sorted(self.latencies)
        k = (len(s) - 1) * (p / 100.0)
        f = int(k)
        c = min(f + 1, len(s) - 1)
        if f == c:
            return s[f]
        return s[f] * (c - k) + s[c] * (k - f)


def parse_records(lines: Iterable[str]) -> Iterator[dict]:
    """Yield dict records from JSONL, skipping blank/invalid lines."""
    for line in lines:
        line = line.strip()
        if not line:
            continue
        try:
            obj = json.loads(line)
        except json.JSONDecodeError:
            continue
        if isinstance(obj, dict):
            yield obj


def aggregate(records: Iterable[dict]) -> dict:
    """Group latency samples by endpoint."""
    stats = defaultdict(EndpointStats)
    for rec in records:
        endpoint = rec.get("endpoint") or rec.get("path") or rec.get("route")
        latency = rec.get("latency_ms") or rec.get("latency") or rec.get("duration_ms")
        if endpoint is None or latency is None:
            continue
        try:
            latency = float(latency)
        except (TypeError, ValueError):
            continue
        stats[endpoint].add(latency)
    return dict(stats)


def format_table(stats: dict, percentiles: list) -> str:
    """Render an aligned ASCII table."""
    headers = ["endpoint", "count"] + [f"p{p:g}" for p in percentiles]
    rows = []
    for endpoint in sorted(stats):
        s = stats[endpoint]
        row = [endpoint, str(len(s.latencies))]
        row += [f"{s.percentile(p):.2f}" for p in percentiles]
        rows.append(row)
    widths = [max([len(h)] + [len(r[i]) for r in rows])
              for i, h in enumerate(headers)]
    lines = ["  ".join(h.ljust(w) for h, w in zip(headers, widths))]
    lines.append("  ".join("-" * w for w in widths))
    for row in rows:
        lines.append("  ".join(c.ljust(w) for c, w in zip(row, widths)))
    return "\n".join(lines)


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("logfile", help="Path to JSONL log file ('-' for stdin)")
    parser.add_argument("-p", "--percentiles", default="50,90,95,99",
                        help="Comma-separated percentiles (default: 50,90,95,99)")
    args = parser.parse_args(argv)

    try:
        percentiles = [float(p) for p in args.percentiles.split(",") if p.strip()]
    except ValueError:
        parser.error("invalid percentiles")
    if not percentiles:
        parser.error("no percentiles given")

    if args.logfile == "-":
        lines = sys.stdin
        close = False
    else:
        lines = open(args.logfile, "r", encoding="utf-8")
        close = True

    try:
        stats = aggregate(parse_records(lines))
    finally:
        if close:
            lines.close()

    if not stats:
        print("No records found.", file=sys.stderr)
        return 1

    print(format_table(stats, percentiles))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
