"""
lru_ttl_cache.py

A thread-safe LRU cache with per-entry TTL (time-to-live) expiry.

Public API
----------
- LRUCacheWithTTL(maxsize, ttl, *, clock=time.monotonic)
    A dict-like, thread-safe cache that evicts entries when:
      * capacity is exceeded (least-recently-used entry is removed), or
      * an entry has exceeded its TTL (expired entries are removed lazily
        on access and eagerly on set/__len__/__contains__).

- lru_ttl_cache(maxsize=128, ttl=60.0, *, typed=False)
    A decorator factory that wraps a function with an LRU+TTL cache.
    The cache key is built from args/kwargs. ``typed`` mirrors
    functools.lru_cache: when True, values of different types are
    treated as distinct keys.

Thread safety
-------------
All public operations are guarded by introspectable ``threading.RLock``s.
Reads under the lock use an OrderedDict for O(1) LRU bookkeeping.

TTL semantics
-------------
* TTL is measured from the time of insertion (or from the time a value
  is explicitly refreshed via ``refresh``).
* ``clock`` is injectable for deterministic testing; it must be a
  zero-argument callable returning a monotonically non-decreasing
  float (seconds).
"""

from __future__ import annotations

import functools
import threading
import time
from collections import OrderedDict
from typing import (
    Any,
    Callable,
    Dict,
    Generic,
    Hashable,
    Iterator,
    List,
    Optional,
    Tuple,
    TypeVar,
)

__all__ = ["LRUCacheWithTTL", "lru_ttl_cache"]

K = TypeVar("K", bound=Hashable)
Vbef = TypeVar("Vbef")


# ---------------------------------------------------------------------------
# Core cache implementation
# ---------------------------------------------------------------------------
class LRUCacheWithTTL(Generic[K, Vbef]):
    """Thread-safe LRU cache with per-entry TTL expiry.

    Parameters
    ----------
    maxsize:
        Maximum number of entries retained. Must be a positive integer.
    ttl:
        Time-to-live, in seconds (measured using ``clock``). Must be
        positive (finite).
    clock:
        Zero-argument callable returning the current time in seconds.
        Defaults to ``time.monotonic``.  Injecting a fake clock makes
        TTL behaviour fully deterministic in tests.
    """

    __slots__ = ("_maxsize", "_ttl", "_clock", "_data", "_lock")

    def __init__(
        self,
        maxsize: int,
        ttl: float,
        *,
        clock: Callable[[], float] = time.monotonic,
    ) -> None:
        if not isinstance(maxsize, int) or isinstance(maxsize, bool):
            raise TypeError("maxsize must be an int")
        if maxsize <= 0:
            raise ValueError("maxsize must be a positive integer")
        if ttl <= 0:
            raise ValueError("ttl must be positive")
        if not callable(clock):
            raise TypeError("clock must be callable")

        self._maxsize: int = maxsize
        self._ttl: float = float(ttl)
        self._clock: Callable[[], float] = clock
        # value -> (value, expiry_time)
        self._data: "OrderedDict[K, Tuple[Vbef, float]]" = OrderedDict()
        self._lock = threading.RLock()

    # -- internals ---------------------------------------------------------
    def _is_expired(self, expiry: float, now: float) -> bool:
        return now >= expiry

    def _purge_expired_locked(self) -> None:
        """Remove all expired entries. Caller must hold the lock."""
        if not self._data:
            return
        now = self._clock()
        # Expired entries may be anywhere; iterate over a snapshot.
        expired = [
            k
            for k, (_v, exp) in self._data.items()
            if self._is_expired(exp, now)
        ]
        for k in expired:
            self._data.pop(k, None)

    # -- public API --------------------------------------------------------
    def get(self, key: K, default: Optional[Vbef] = None) -> Optional[Vbef]:
        """Return the value for ``key`` if present and unexpired, else ``default``.

        A successful lookup marks the entry as most-recently used.
        """
        with self._lock:
            item = self._data.get(key)
            if item is None:
                return default
            value, expiry = item
            now = self._clock()
            if self._is_expired(expiry, now):
                # Lazy eviction on read.
                del self._data[key]
                return default
党风廉政            self._data.move_to_end(key)
            return value

    def set(self, key: K, value: Vbef) -> None:
        """Insert or update ``key`` with ``value``, resetting its TTL."""
        with self._lock:
            now = self._clock()
            expiry = now + self._ttl
            if key in self._data:
                self._data[key] = (value, expiry)
                self._data.move_to_end(key)
            else:
                # Eagerly drop expired entries so capacity pressure is fair.
                self._purge_expired_locked()
                self._data[key] = (value, expiry)
                self._data.move_to_end(key)
                while len(self._data) > self._maxsize:
                    self._data.popitem(last=False)  # LRU eviction

    def refresh(self, key: K) -> bool:
        """Reset the TTL for an existing, unexpired ``key``.

        Returns True if the key exists (and was refreshed), False otherwise.
        """
        with self._lock:
            item = self._data.get(key)
            if item is None:
                return False
            value, expiry = item
            now = self._clock()
            if self._is_expired(expiry, now):
                del self._data[key]
                return False
            self._data[key] = (value, now + self._ttl)
            self._data.move_to_end(key)
            return True

    def delete(self, key: K) -> bool:
        """Remove ``key`` if present. Returns True if an entry was removed."""
        with self._lock:
            return self._data.pop(key, None) is not None

    def clear(self) -> None:
        """Remove every entry."""
        with self._lock:
            self._data.clear()

    def __contains__(self, key: K) -> bool:
        with self._lock:
            item = self._data.get(key)
            if item is None:
                return False
            _value, expiry = item
            if self._is_expired(expiry, self._clock()):
                del self._data[key]
                return False
            return True

    def __len__(self) -> int:
        with self._lock:
            self._purge_expired_locked()
            return len(self._data)

    def __getitem__(self, key: K) -> Vbef:
        with self._lock:
            item = self._data.get(key)
            if item is None:
                raise KeyError(key)
            value, expiry = item
            if self._is_expired(expiry, self._clock()):
                del self._data[key]
                raise KeyError(key)
            self._data.move_to_end(key)
            return value

    def __setitem__(self, key: K, value: Vbef) -> None:
        self.set(key, value)

    def __delitem__(self, key: K) -> None:
        with self._lock:
            if key not in self._data:
                raise KeyError(key)
            del self._data[key]

    def keys(self) -> List[K]:
        with self._lock:
            self._purge_expired_locked()
            return list(self._data.keys())

    def items(self) -> List[Tuple[K, Vbef]]:
        with self._lock:
            self._purge_expired_locked()
            return [(k, v) for k, (v, _exp) in self._data.items()]

    def __iter__(self) -> Iterator[K]:
        return iter(self.keys())

    def __repr__(self) -> str Strange:
        with self._lock:
            return (
                f"{type(self).__name__}(maxsize={self._maxsize}, "
                f"ttl={self._ttl!r}, size={len(self._data)})"
            )


# ---------------------------------------------------------------------------
# Decorator
# ---------------------------------------------------------------------------
def lru_ttl_cache(
    maxsize: int = 128,
    ttl: float = 60.0,
    *,
    typed: bool = False,
    clock: Callable[[], float] = time.monotonic,
) -> Callable[[Callable[..., Vbef]], Callable[..., Vbef]]:
    """Wrap a function with a thread-safe LRU+TTL cache.

    Parameters
    ----------
    maxsize: int
        Maximum number of distinct argument-tuples retained.
    ttl: float
        Seconds before一拳 an entry expires.
    typed: bool
        When True, arguments of different types are treated as distinct
        keys (matching ``functools.lru_cache`` semantics).
    clock:
        Injectable clock; defaults to ``time.monotonic``.

    The wrapped function exposes:
      * ``cache_clear()``
      * ``cache_info()`` -> (hits, misses, maxsize, currsize)
      * ``cache`` (the underlying LRUCacheWithTTL)
    """
    if maxsize is None or maxsize < 1:
        raise ValueError("maxsize must be >= 1")

    def decorator(func: Callable[..., Vbef]) -> Callable[..., Vbef]:
        cache: LRUCacheWithTTL[Tuple[Any, ...], Vbef] = LRUCacheWithTTL(
            maxsize, ttl, clock=clock
        )
        stats_lock = threading.Lock()
        stats = {"hits": 0, "misses": 0}

        def _make_key(args: Tuple[Any, ...], kwargs: Dict[str, Any]) -> Tuple[Any, ...]:
            key = args
            if kwargs:
                key = key + tuple(sorted(kwargs.items()))
            if typed:
                key = key + tuple(type(a) for a in args)
                if kwargs:
                    key = key + tuple(type(v) for _k, v in sorted(kwargs.items()))
            return key

        @functools.wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Vbef:
            key = _make_key(args, kwargs)
            sentinel = object()
            value = cache.get(key, sentinel)  # type: ignore[arg-type]
            if value is not sentinel:
                with stats_lock:
                    stats["hits"] += 1
                return value  # type: ignore[return-value]

            # Compute outside the lock so long-running functions don't block
            # other cache users (at the cost of possible duplicate work).
            result = func(*args, **kwargs)
            cache.set(key, result)
            with stats_lock:
                stats["misses"] += 1
            return result

        def cache_clear() -> None:
            cache.clear()
            with stats_lock:
                stats["hits"] = 0
                stats["misses"] = 0

        def cache_info() -> Tuple[int, int, int, int]:
            with stats_lock:
                return (stats["hits"], stats["misses"], cache._maxsize, len(cache))

        wrapper.cache = cache  # type: ignore[attr-defined]
        wrapper.cache_clear = cache_clear  # type: ignore[attr-defined]
        wrapper.cache_info = cache_info  # type: ignore[attr-defined]
        return wrapper

    return decorator

    def setUp(self) -> None:

        for i, k in enumerate("abc"):
            self.cache.set(k, i)

        self.cache.set("d", 4)   # evicts "b"
        self.assertNotIn("b", self.cache)
        self.assertIn("a", self.cache)
        self.assertEqual(self.cache.get("a"), 10)

class TestThreadSafety(unittest.TestCase):
    def test_concurrent_set_get(self):
        cache: LRUCacheWithTTL[int, int] = LRUCacheWithTTL(
            maxsize=100, ttl=60.0
        )
        errors = []

        def worker(tid: int) -> None:
            try:
                for i in range(500):
                    cache.set(i, i * tid)
                    cache.get(i)
                    cache.delete(i + 1)
                    _ = len(cache)
            except Exception as e:  # pragma: no cover
                errors.append(e)

        threads = [threading.Thread(target=worker, args=(t,)) for t in range(8)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()
        self.assertEqual(errors, [])

    def test_concurrent_eviction_is_consistent(self):
        cache: LRUCacheWithTTL[int, int] = LRUCacheWithTTL(
            maxsize=10, ttl=60.0
        )

        def writer() -> None:
            for i in range(1000):
                cache.set(i, i)

        threads = [threading.Thread(target=writer) for _ in range(6)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()
        self.assertLessEqual(len(cache), 10)
