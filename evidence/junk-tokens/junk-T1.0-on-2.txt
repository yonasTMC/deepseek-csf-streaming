    start = time.monotonic()
    last_status: int | None = None
    last_error: str | None = None
    attempts = 0

    for attempt in range(1, retries + 2):
        attempts = attempt
        try:
            status = await _http_get_status(url, timeout)
            last_status = status

            if status in expected:
                return CheckResult(
                    url=url,
                    healthy=True,
                    status=status,
                    attempts=attempts,
                    elapsed=time.monotonic() - start,
                    error=None,
                )

            last_error = f"unexpected status {status}"
        except Exception as exc:
            last_status = None
            last_error = f"{type(exc).__name__}: {exc}"

        if attempt <= retries:
            delay = min(
                max_delay,
                base_delay * (backoff_factor ** (attempt - 1)),
            )
            if jitter:
                delay += random.uniform(0.0, jitter)
            if delay > 0:
                await asyncio.sleep(delay)

    return CheckResult(
        url=url,
        healthy=False,
        status=last_status,
        attempts=attempts,
        elapsed=time.monotonic() - start,
        error=last_error,
    )


async def check_urls(
    urls: Sequence[str],
    *,
    retries: int = 3,
    base_delay: float = 0.1,
    max_delay: float = 5.0,
    backoff_factor: float = 2.0,
    timeout: float = 5.0,
    expected_status: int | Iterable[int] = range(200, 300),
    jitter: float = 0.0,
) -> HealthReport:
    """Check all URLs concurrently and return an aggregate report."""
    expected = _normalize_expected_status(expected_status)
    start = time.monotonic()

    results = await asyncio.gather(
        *(
            check_url(
                url,
                retries=retries,
                base_delay=base_delay,
                max_delay=max_delay,
                backoff_factor=backoff_factor,
                timeout=timeout,
                expected_status=expected,
                jitter=jitter,
            )
            for url in urls
        )
    )

    return HealthReport(
        results=tuple(results),
        elapsed=time.monotonic() - start,
    )

async def _start_server(statuses: list[int]):
    requests = 0

    async def handler(
        reader: asyncio.StreamReader,
        writer: asyncio.StreamWriter,
    ) -> None:
        nonlocal requests

        try:
            await reader.readuntil(b"\r\n\r\n")
        except (asyncio.IncompleteReadError, ConnectionError):
            pass

        status = statuses[min(requests, len(statuses) - 1)]
        requests += 1

        body = b"ok" if 200 <= status < 300 else b"fail"
        response = (
            f"HTTP/1.1 {status} Test\r\n"
            f"Content-Length: {len(body)}\r\n"
            "Connection: close\r\n"
            "\r\n"
        ).encode("ascii") + body

        writer.write(response)
        try:
            await writer.drain()
        finally:
            writer.close()
            try:
                await writer.wait_closed()
            except Exception:
                pass

    server = await asyncio.start_server(handler, "127.0.0.1", 0)
    port = server.sockets[0].getsockname()[1]
    return server, port, lambda: requests
