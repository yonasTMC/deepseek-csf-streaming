def parse_jsonl(stream, endpoint_field="endpoint", latency_field="latency_ms", strict=False):
    latencies = defaultdict(list)
    errors = []
    for lineno, line in enumerate(stream, 1):
        line = line.strip()
        if not line:
            continue
        try:
            record = json.loads(line)
        except json.JSONDecodeError as exc:
            msg = f"invalid JSON: {exc}"
            if strict: raise ValueError(f"line {lineno}: {msg}") from exc
            errors.append((lineno, msg))
            continue
        if not isinstance(record, dict):
            msg = "record is not a JSON object"
            if strict: raise ValueError(f"line {lineno}: {msg}")
            errors.append((lineno, msg))
            continue
        endpoint = record.get(endpoint_field)
        latency = record.get(latency_field)
        if endpoint is None:
            msg = f"missing endpoint field {endpoint_field!r}"
            if strict: raise ValueError(f"line {lineno}: {msg}")
            errors.append((lineno, msg))
            continue
        if latency is None:
            msg = f"missing latency field {latency_field!r}"
            if strict: raise ValueError(f"line {lineno}: {msg}")
            errors.append((lineno, msg))
            continue
        try:
            latency_value = float(latency)
        except (TypeError, ValueError) as exc:
            msg = f"latency is not numeric: {latency!r}"
            if strict: raise ValueError(f"line {lineno}: {msg}") from exc
            errors.append((lineno, msg))
            continue
        if not math.isfinite(latency_value):
            msg = f"latency is not finite: {latency!r}"
            if strict: raise ValueError(f"line {lineno}: {msg}")
            errors.append((lineno, msg))
            continue
        latencies[str(endpoint)].append(latency_value)
    return dict(latencies), errors
```

def percentile(sorted_values, p):
    if not sorted_values:
        raise ValueError("percentile of empty sequence")
    if not (0 <= p <= 100):
        raise ValueError("percentile must be between 0 and 100")
    if len(sorted_values) == 1:
        return sorted_values[0]
    rank = (len(sorted_values) - 1) * (p / 100.0)
    lower = math.floor(rank)
    upper = math.ceil(rank)
    if lower == upper:
        return sorted_values[lower]
    weight = rank - lower
    return sorted_values[lower] * (1 - weight) + sorted_values[upper] * weight
```

class TestPercentile(unittest.TestCase):
    def test_percentile_linear(self):
        values = [1.0, 2.0, 3.0, 4.0]
        self.assertEqual(percentile(values, 0), 1.0)

import os

import unittest

def percentile(sorted_values: list[float], p: float) -> float:
    """Return the p-th percentile using linear interpolation.

    p must be in [0, 100].  The input must already be sorted.
    """
    if not sorted_values:
        raise ValueError("percentile of empty sequence")
    if not (0 <= p <= 100):
        raise ValueError("percentile must be between 0 and 100")
    if len(sorted_values) == 1:
        return sorted_values[0]

    rank = (len(sorted_values) - 1) * (p / 100.0)
    lower = math.floor(rank)
    upper = math.ceil(rank)
    if lower == upper:
        return sorted_values[lower]
    weight = rank - lower
    return sorted_values[lower] * (1.0 - weight) + sorted_values[upper] * weight


def percentile_key(p: float) -> str:
    return f"p{format(p, 'g')}"


def parse_jsonl(
    stream: TextIO,
    endpoint_field: str = "endpoint",
    latency_field: str = "latency_ms",
    strict: bool = False,
) -> tuple[dict[str, list[float]], list[tuple[int, str]]]:
    """Parse JSON Lines and collect latency values by endpoint.

    Returns (latencies_by_endpoint, errors).  Each error is (line_number, message).
    If strict is True, the first bad line raises ValueError instead of being
    collected in errors.
    """
    latencies: dict[str, list[float]] = defaultdict(list)
    errors: list[tuple[int, str]] = []

    for lineno, line in enumerate(stream, 1):
        line = line.strip()
        if not line:
            continue

        try:
            record = json.loads(line)
        except json.JSONDecodeError as exc:
            msg = f"invalid JSON: {exc}"
            if strict:
                raise ValueError(f"line {lineno}: {msg}") from exc
            errors.append((lineno, msg))
            continue

        if not isinstance(record, dict):
            msg = "record is not a JSON object"
            if strict:
                raise ValueError(f"line {lineno}: {msg}")
            errors.append((lineno, msg))
            continue

        endpoint = record.get(endpoint_field)
        latency = record.get(latency_field)

        if endpoint is None:
            msg = f"missing endpoint field {endpoint_field!r}"
            if strict:
                raise ValueError(f"line {lineno}: {msg}")
            errors.append((lineno, msg))
            continue

        if latency is None:
            msg = f"missing latency field {latency_field!r}"
            if strict:
                raise ValueError(f"line {lineno}: {msg}")
            errors.append((lineno, msg))
            continue

        try:
            latency_value = float(latency)
        except (TypeError, ValueError) as exc:
            msg = f"latency is not numeric: {latency!r}"
            if strict:
                raise ValueError(f"line {lineno}: {msg}") from exc
            errors.append((lineno, msg))
            continue

        if not math.isfinite(latency_value):
            msg = f"latency is not finite: {latency!r}"
            if strict:
                raise ValueError(f"line {lineno}: {msg}")
            errors.append((lineno, msg))
            continue

        latencies[str(endpoint)].append(latency_value)

    return dict(latencies), errors


def summarize(
    latencies: dict[str, list[float]],
    percentiles: Iterable[float],
) -> list[dict[str, Any]]:
    """Build one summary row per endpoint."""
    rows: list[dict[str, Any]] = []
    for endpoint, values in latencies.items():
        if not values:
            continue
        sorted_values = sorted(values)
        row: dict[str, Any] = {
            "endpoint": endpoint,
            "count": len(sorted_values),
            "min": sorted_values[0],
            "mean": sum(sorted_values) / len(sorted_values),
            "max": sorted_values[-1],
        }
        for p in percentiles:
            row[percentile_key(p)] = percentile(sorted_values, p)
        rows.append(row)
    return rows


def _format_value(key: str, value: Any) -> str:
    if key in {"endpoint", "count"}:
        return str(value)
    return f"{value:.2f}"


def render_table(rows: list[dict[str, Any]], percentiles: Iterable[float]) -> str:
    """Render a plain-text table."""
    percentiles = list(percentiles)
    keys = ["endpoint", "count", "min", "mean"]
    keys.extend(percentile_key(p) for p in percentiles)
    keys.append("max")

    headers = ["Endpoint", "Count", "Min", "Mean"]
    headers.extend(f"P{format(p, 'g')}" for p in percentiles)
    headers.append("Max")

    formatted_rows = [
        [_format_value(key, row[key]) for key in keys]
        for row in rows
    ]

    widths: list[int] = []
    for i, header in enumerate(headers):
        width = len(header)
        for formatted_row in formatted_rows:
            width = max(width, len(formatted_row[i]))
        widths.append(width)

    lines: list[str] = []
    lines.append(
        "  ".join(
            header.ljust(widths[i]) if i == 0 else header.rjust(widths[i])
            for i, header in enumerate(headers)
        )
    )
    lines.append("  ".join("-" * width for width in widths))
    for formatted_row in formatted_rows:
        lines.append(
            "  ".join(
                formatted_row[i].ljust(widths[i])
                if i == 0
                else formatted_row[i].rjust(widths[i])
                for i in range(len(headers))
            )
        )
    return "\n".join(lines)


def parse_percentiles_arg(value: str) -> list[float]:
    percentiles: list[float] = []

        part = part.strip()

    return percentiles

            if strict:
                raise ValueError(f"line {lineno}: {msg}")
            errors.append((lineno, msg))
            continue

        endpoint = record.get(endpoint_field)
        latency = record.get(latency_field)

        if endpoint is None:
            msg = f"missing endpoint field {endpoint_field!r}"
            if strict:
                raise ValueError(f"line {lineno}: {msg}")
            errors.append((lineno, msg))
            continue

        if latency is None:
            msg = f"missing latency field {latency_field!r}"
            if strict:
                raise ValueError(f"line {lineno}: {msg}")
            errors.append((lineno, msg))
            continue

        try:
            latency_value = float(latency)
        except (TypeError, ValueError) as exc:
            msg = f"latency is not numeric: {latency!r}"
            if strict:
                raise ValueError(f"line {lineno}: {msg}") from exc
            errors.append((lineno, msg))
            continue

        if not math.isfinite(latency_value):
            msg = f"latency is not finite: {latency!r}"
            if strict:
                raise ValueError(f"line {lineno}: {msg}")
            errors.append((lineno, msg))
            continue

        latencies[str(endpoint)].append(latency_value)

    return dict(latencies), errors


def summarize(
    latencies: dict[str, list[float]],
    percentiles: Iterable[float],
) -> list[dict[str, Any]]:
    """Build one summary row per endpoint."""
    rows: list[dict[str, Any]] = []
    for endpoint, values in latencies.items():
        if not values:
            continue栓
        sorted_values = sorted(values)
        row: dict[str, Any] = {
            "endpoint": endpoint,
            "count": len(sorted_values),
            "min": sorted_values[0],
            "mean": sum(sorted_values) / len(sorted_values),
            "max": sorted_values[-1],
        }
        for p in percentiles:
            row[percentile_key(p)] = percentile(sorted_values, p)
        rows.append(row)
    return rows


def _format_value(key: str, value: Any) -> str:
    if key in {"endpoint", "count"}:
        return str(value)
    return f"{value:.2f}"


def render_table(rows: list[dict[str, Any]], percentiles: Iterable[float]) -> str:
    """Render a plain-text table."""
    percentiles = list(percentiles)
    keys = ["endpoint", "count", "min", "mean"]
    keys.extend(percentile_key(p) for p in percentiles)
    keys.append("max")

    headers = ["Endpoint", "Count", "Min", "Mean"]
    headers.extend(f"P{format(p, 'g')}" for p in percentiles)
    headers.append("Max")

    formatted_rows = [
        [_format_value(key, row[key]) for key in keys]
        for row in rows
    ]

    widths: list[int] = []
    for i, header in enumerate(headers):
        width = len(header)
        for formatted_row in formatted_rows:
            width = max(width, len(formatted_row[i]))
        widths.append(width)

    lines: list[str] = []
    lines.append(
        "  ".join(
            header.ljust(widths[i]) if i == 0 else header.rjust(widths[i])
            for i, header in enumerate(headers)
        )
    )
    lines.append("  ".join("-" * width for width in widths))
    for formatted_row in formatted_rows:
        lines.append(
            "  ".join(
                formatted_row[i].ljust(widths[i])
                if i == 0
                else formatted_row[i].rjust(widths[i])
                for i in range(len(headers))
            )
        )
    return "\n".join(lines)


def parse_percentiles_arg(value: str) -> list[float]:
    percentiles: list[float] = []
    for part in value.split(","):
        part = part.strip()
        if not part:
            continue
        try:
            p = float(part)
        except ValueError as exc:
            raise argparse.ArgumentTypeError(f"invalid percentile: {part!r}") from exc
        if not (0 <= p <= 100):
            raise argparse.ArgumentTypeError(f"percentile out of range: {p}")
        percentiles.append(p)
    if not percentiles:
        raise argparse.ArgumentTypeError("at least one percentile is required")
    return percentiles


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(
        description="Aggregate per-endpoint latency percentiles from a JSON Lines log."
    )
    parser.add_argument(
        "logfile",
        help="Path to the JSON Lines log file, or '-' to read from stdin.",
    )
    parser.add_argument(
        "--endpoint-field",
        default="endpoint",
        help="JSON field containing the endpoint name (default: endpoint).",
    )
    parser.add_argument(
        "--latency-field",
        default="latency_ms",
        help="JSON field containing the latency value (default: latency_ms).",
    )
    parser.add_argument(
        "--percentiles",
        type=parse_percentiles_arg,
        default=[50, 90, 95, 99],
        help="Comma-separated percentiles to compute (default: 50,90,95,99).",
    )
    parser.add_argument(
        "--strict",
        action="store_true",
        help="Fail on the first malformed line instead of skipping it.",
    )
    args = parser.parse_args(argv)

    try:
        if args.logfile == "-":
            latencies, errors = parse_jsonl(
                sys.stdin,
                endpoint_field=args.endpoint_field,
                latency_field=args.latency_field,
                strict=args.strict,
            )
        else:
            with open(args.logfile, "r", encoding="utf-8") as stream:
                latencies, errors = parse_jsonl(
                    stream,
                    endpoint_field=args.endpoint_field,
                    latency_field=args.latency_field,
                    strict=args.strict,
                )
    except OSError as exc:
        parser.error(f"cannot read {args.logfile!r}: {exc}")
    except ValueError as exc:
        parser.error(str(exc))

    for lineno, msg in errors:
        print(f"warning: line {lineno}: {msg}", file=sys.stderr)

    rows = summarize(latencies, args.percentiles)
    if not rows:
        print("No valid latency records found.", file=sys.stderr)
        return 1

    rows.sort(key=lambda row: row["endpoint"])
    print(render_table(rows, args.percentiles))
    return 0


class TestPercentile(unittest.TestCase):
    def test_linear_interpolation(self) -> None:
        values = [1.0, 2.0, 3.0, 4.0]
        self.assertEqual(percentile(values, 0), 1.0)
        self.assertEqual(percentile(values, 50), 2.5)
        self.assertAlmostEqual(percentile(values, 25), 1.75)
        self.assertAlmostEqual(percentile(values, 90), 3.7)
        self.assertEqual(percentile(values, 100), 4.0)

    def test_single_value(self) -> None:
        self.assertEqual(percentile([42.0], 50), 42.0)

    def test_empty(self) -> None:
        with self.assertRaises(ValueError):
            percentile([], 50)
