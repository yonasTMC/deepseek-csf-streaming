from __future__ import annotations

import asyncio
import ssl
import sys
import time
from dataclasses import dataclass
from typing import Iterable, Sequence
from urllib.parse import urlsplit

__all__ = [
    "HealthCheckConfig",
    "HealthCheckResult",
    "HealthChecker",
    "summarize",
]


@dataclass(frozen=True)
class HealthCheckConfig:
    timeout: float = 5.0
    retries: int = 3
    backoff_base: float = 0.1
    backoff_factor: float = 2.0
    max_backoff: float = 5.0
    expected_status: int = 200
    user_agent: str = "asyncio-health-checker/1.0"


@dataclass
class HealthCheckResult:
    url: str
    ok: bool
    status: int | None
    attempts: int
    elapsed: float
    error: str | None = None


class HealthChecker:
    def __init__(self, config: HealthCheckConfig | None = None) -> None:
        self.config = config or HealthCheckConfig()

    async def check(self, url: str) -> HealthCheckResult:
        start = time.monotonic()
        last_status: int | None = None
        last_error: str | None = None
        attempts = 0

        for attempt in range(1, self.config.retries + 2):
            attempts = attempt
            try:
                status = await asyncio.wait_for(
                    self._request(url),
                    timeout=self.config.timeout,
                )
                last_status = status
                if status == self.config.expected_status:
                    return HealthCheckResult(
                        url=url,
                        ok=True,
                        status=status,
                        attempts=attempts,
                        elapsed=time.monotonic() - start,
                    )
                last_error = f"unexpected status {status}"
            except Exception as exc:
                last_error = f"{type(exc).__name__}: {exc}"

            if attempt <= self.config.retries:
                delay = min(
                    self.config.backoff_base
                    * (self.config.backoff_factor ** (attempt - 1)),
                    self.config.max_backoff,
                )
                await asyncio.sleep(delay)

        return HealthCheckResult(
            url=url,
            ok=False,
            status=last_status,
            attempts=attempts,
            elapsed=time.monotonic() - start,
            error=last_error,
        )

    async def check_many(self, urls: Sequence[str]) -> list[HealthCheckResult]:
        return await asyncio.gather(*(self.check(url) for url in urls))

    async def _request(self, url: str) -> int:
        parts = urlsplit(url)
        if parts.scheme not in {"http", "https"}:
            raise ValueError(f"unsupported URL scheme: {parts.scheme!r}")

        host = parts.hostname
        if not host:
            raise ValueError(f"URL has no host: {url!r}")

        default_port = 443 if parts.scheme == "https" else 80
        port = parts.port or default_port
        path = parts.path or "/"
        if parts.query:
            path = f"{path}?{parts.query}"

        host_header = host
        if ":" in host and not host.startswith("["):
            host_header = f"[{host}]"
        if port != default_port:
            host_header = f"{host_header}:{port}"

        ssl_context = ssl.create_default_context() if parts.scheme == "https" else None
        reader, writer = await asyncio.open_connection(host, port, ssl=ssl_context)
        try:
            request = (
                f"GET {path} HTTP/1.1\r\n"
                f"Host: {host_header}\r\n"
                f"User-Agent: {self.config.user_agent}\r\n"
                f"Connection: close\r\n"
                f"\r\n"
            )
            writer.write(request.encode("ascii"))
            await writer.drain()

            status_line = await reader.readline()
            if not status_line:
                raise ConnectionError("empty response")

            status_parts = status_line.decode("latin-1").split()
            if len(status_parts) < 2 or not status_parts[0].startswith("HTTP/"):
                raise ValueError(f"invalid status line: {status_line!r}")

            return int(status_parts[1])
        finally:
            writer.close()
            try:
                await writer.wait_closed()
            except Exception:
                pass


def summarize(results: Iterable[HealthCheckResult]) -> str:
    results = list(results)
    total = len(results)
    healthy = sum(1 for result in results if result.ok)
    unhealthy = total - healthy

    lines = [
