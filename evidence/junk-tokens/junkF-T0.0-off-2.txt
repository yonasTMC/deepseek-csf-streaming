```python
# health_checker.py
"""
Asyncio HTTP Health Checker with retries, exponential backoff, and summary reporting.
"""

import asyncio
import time
import random
from dataclasses import dataclass, field
from typing import List, Optional, Dict, Any
from enum import Enum

try:
    import aiohttp
except ImportError:
    aiohttp = None


class HealthStatus(Enum):
    HEALTHY = "healthy"
    UNHEALTHY = "unhealthy"
    DEGRADED = "degraded"
    UNKNOWN = "unknown"


@dataclass
class EndpointConfig:
    """Configuration for a single endpoint to check."""
    url: str
    name: Optional[str] = None
    method: str = "GET"
    expected_status: int = 200
    timeout: float = 5.0
    max_retries: int = 3
    base_backoff: float = 0.5
    max_backoff: float = 10.0
    jitter: bool = True
    headers: Dict[str, str] = field(default_factory=dict)

    def __post_init__(self):
        if self.name is None:
            self.name = self.url


@dataclass
class CheckResult:
    """Result of a single endpoint health check."""
    name: str
    url: str
    status: HealthStatus
    attempts: int
    total_time: float
    status_code: Optional[int] = None
    error: Optional[str] = None
    attempt_times: List[float] = field(default_factory=list)


@dataclass
class HealthReport:
    """Aggregated health report for all endpoints."""
    results: List[CheckResult]
    start_time: float
    end_time: float

    @property
    def duration(self) -> float:
        return self.end_time - self.start_time

    @property
    def overall_status(self) -> HealthStatus:
        if not self.results:
            return HealthStatus.UNKNOWN
        statuses = {r.status for r in self.results}
        if statuses == {HealthStatus.HEALTHY}:
            return HealthStatus.HEALTHY
        if HealthStatus.HEALTHY in statuses:
            return HealthStatus.DEGRADED
        return HealthStatus.UNHEALTHY

    def summary(self) -> str:
        """Generate a human-readable summary report."""
        lines = []
        lines.append("=" * 70)
        lines.append("HEALTH CHECK REPORT")
        lines.append("=" * 70)
        lines.append(f"Overall Status: {self.overall_status.value.upper()}")
        lines.append(f"Total Duration: {self.duration:.3f}s")
        lines.append(f"Endpoints Checked: {len(self.results)}")
        lines.append("-" * 70)

        counts = {s: 0 for s in HealthStatus}
        for r in self.results:
            counts[r.status] += 1

        lines.append(
            f"  Healthy: {counts[HealthStatus.HEALTHY]}  |  "
            f"Degraded: {counts[HealthStatus.DEGRADED]}  |  "
            f"Unhealthy: {counts[HealthStatus.UNHEALTHY]}  |  "
            f"Unknown: {counts[HealthStatus.UNKNOWN]}"
        )
        lines.append("-" * 70)

        for r in self.results:
            lines.append(f"\n[{r.status.value.upper()}] {r.name}")
            lines.append(f"  URL: {r.url}")
            lines.append(f"  Attempts: {r.attempts}")
            lines.append(f"  Total Time: {r.total_time:.3f}s")
            if r.status_code is not None:
                lines.append(f"  Status Code: {r.status_code}")
            if r.error:
                lines.append(f"  Error: {r.error}")
            if r.attempt_times:
                times_str = ", ".join(f"{t:.3f}s" for t in r.attempt_times)
                lines.append(f"  Attempt Times: [{times_str}]")

        lines.append("=" * 70)
        return "\n".join(lines)


class HealthChecker:
    """Asyncio-based HTTP health checker."""

    def __init__(
        self,
        endpoints: List[EndpointConfig],
        max_concurrency: int = 10,
        session_factory=None,
    ):
        self.endpoints = endpoints
        self.max_concurrency = max_concurrency
        self._session_factory = session_factory
        self._semaphore: Optional[asyncio.Semaphore] = None

    def _compute_backoff(self, attempt: int, config: EndpointConfig) -> float:
        """Compute exponential backoff with optional jitter."""
        delay = min(config.base_backoff * (2 ** attempt), config.max_backoff)
        if config.jitter:
            delay = delay * (0.5 + random.random())
        return delay

    async def _check_once(
        self, session, config: EndpointConfig
    ) -> tuple:
        """Perform a single HTTP request. Returns (success, status_code, error)."""
        try:
            async with session.request(
                config.method,
                config.url,
                headers=config.headers,
                timeout=aiohttp.ClientTimeout(total=config.timeout),
            ) as response:
                status_code = response.status
                if status_code == config.expected_status:
                    return True, status_code, None
                return False, status_code, f"Unexpected status: {status_code}"
        except asyncio.TimeoutError:
            return False, None, "Timeout"
        except aiohttp.ClientError as e:
            return False, None, f"ClientError: {e}"
        except Exception as e:
            return False, None, f"Error: {e}"

    async def check_endpoint(
        self, session, config: EndpointConfig
    ) -> CheckResult:
        """Check a single endpoint with retries and exponential backoff."""
        async with self._semaphore:
            start = time.monotonic()
            attempt_times: List[float] = []
            last_status_code: Optional[int] = None
            last_error: Optional[str] = None
            attempts = 0

            for attempt in range(config.max_retries + 1):
                attempts = attempt + 1
                attempt_start = time.monotonic()
                success, status_code, error = await self._check_once(
                    session, config
                )
                attempt_times.append(time.monotonic() - attempt_start)
                last_status_code = status_code
                last_error = error

                if success:
                    return CheckResult(
                        name=config.name,
                        url=config.url,
                        status=HealthStatus.HEALTHY,
                        attempts=attempts,
                        total_time=time.monotonic() - start,
                        status_code=status_code,
                        attempt_times=attempt_times,
                    )

                if attempt < config.max_retries:
                    delay = self._compute_backoff(attempt, config)
                    await asyncio.sleep(delay)

            return CheckResult(
                name=config.name,
                url=config.url,
                status=HealthStatus.UNHEALTHY,
                attempts=attempts,
                total_time=time.monotonic() - start,
                status_code=last_status_code,
                error=last_error,
                attempt_times=attempt_times,
            )

    async def run(self) -> HealthReport:
        """Run all health checks concurrently and return a report."""
        if aiohttp is None:
            raise RuntimeError("aiohttp is required for HealthChecker")

        self._semaphore = asyncio.Semaphore(self.max_concurrency)
        start = time.monotonic()

        session_factory = self._session_factory or aiohttp.ClientSession
        async with session_factory() as session:
            tasks = [
                self.check_endpoint(session, config)
                for config in self.endpoints
            ]
            results = await asyncio.gather(*tasks)

        end = time.monotonic()
        return HealthReport(
            results=list(results),
            start_time=start,
            end_time=end,
        )


async def main():
    """Example usage."""
    endpoints = [
        EndpointConfig(url="https://httpbin.org/status/200", name="httpbin-200"),
        EndpointConfig(url="https://httpbin.org/status/500", name="httpbin-500"),
        EndpointConfig(
            url="https://httpbin.org/delay/1",
            name="httpbin-delay",
            timeout=2.0,
        ),
    ]
    checker = HealthChecker(endpoints, max_concurrency=5)
    report = await checker.run()
    print(report.summary())


if __name__ == "__main__":
    asyncio.run(main())
```

```python
    @pytest.mark.asyncio
    async def test_run_respects_concurrency(self):
        """Ensure no more than max_concurrency checks run at once."""
        active = 0
        peak = 0
        lock = asyncio.Lock()

        async def fake_check(session, config):
            nonlocal active, peak
            async with lock:
                active += 1
                peak = max(peak, active)
            await asyncio.sleep(0.02)
            async with lock:
                active -= 1
            return CheckResult(
                name=config.name,
                url=config.url,
                status=HealthStatus.HEALTHY,
                attempts=1,
                total_time=0.02,
            )

        endpoints = [
            EndpointConfig(url=f"http://x{i}.com", name=f"x{i}")
            for i in range(10)
        ]
        checker = HealthChecker(endpoints, max_concurrency=3)
        checker.check_endpoint = fake_check

        mock_session = MagicMock()
        mock_session.__aenter__ = AsyncMock(return_value=mock_session)
        mock_session.__aexit__ = AsyncMock(return_value=False)
        checker._session_factory = lambda: mock_session

        await checker.run()
        assert peak <= 3

@pytest.mark.asyncio
async def test_end_to_end_with_local_server():
    """Spin up a real aiohttp server and verify the checker against it."""
    aiohttp = pytest.importorskip("aiohttp")
    from aiohttp import web

    async def ok_handler(request):
        return web.Response(status=200, text="ok")

    async def fail_handler(request):
        return web.Response(status=500, text="fail")

    app = web.Application()
    app.router.add_get("/ok", ok_handler)
    app.router.add_get("/fail", fail_handler)

    runner = web.AppRunner(app)
    await runner.setup()
    site = web.TCPSite(runner, "127.0.0.1", 0)
    await site.start()

    # Discover the bound port
    port = None
    for sock in site._server.sockets:
        port = sock.getsockname()[1]
        break
    assert port is not None

    try:
        endpoints = [
            EndpointConfig(
                url=f"http://127.0.0.1:{port}/ok",
                name="ok",
                max_retries=1,
                base_backoff=0.01,
                jitter=False,
            ),
            EndpointConfig(
                url=f"http://127.0.0.1:{port}/fail",
                name="fail",
                max_retries=1,
                base_backoff=0.01,
                jitter=False,
            ),
        ]
        checker = HealthChecker(endpoints, max_concurrency=2)
        report = await checker.run()

        statuses = {r.name: r.status for r in report.results}
        assert statuses["ok"] == HealthStatus.HEALTHY
        assert statuses["fail"] == HealthStatus.UNHEALTHY
        assert report.overall_status == HealthStatus.DEGRADED
    finally:
        await runner.cleanup()
```

## Notes

- **Retries & backoff**: `_compute_backoff` uses `base_backoff * 2**attempt`, capped at `max_backoff`, with optional jitter (`0.5x`–`1.5x`) to avoid thundering-herd effects.
- **Concurrency**: A single `asyncio.Semaphore` bounds in-flight requests across all endpoints.
- **Session injection**: `session_factory` allows tests to inject a mock session without patching `aiohttp`.
- **Report**: `HealthReport.summary()` produces a readable text report; `overall_status` aggregates per-endpoint results (`HEALTHY` / `DEGRADED` / `UNHEALTHY` / `UNKNOWN`).
- **Tests**: Cover config defaults, backoff math (growth, cap, jitter bounds), retry behavior (success, recovery, exhaustion), error handling (timeout, generic exception), report aggregation, concurrency limits, and a real end-to-end test against a local `aiohttp` server.
